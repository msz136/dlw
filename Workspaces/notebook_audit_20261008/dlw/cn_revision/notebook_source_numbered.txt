0001 ### 1.1　孤子精确解
0002 
0003 $x,y$ 为空间坐标，$t$ 为时间。由表中的 $p_i,q_i$ 及 $a=2$，确定
0004 
0005 $$S_i=p_i+q_i,\qquad \omega_i=q_i^2-p_i^2,\qquad
0006 \ell_i=\frac1{p_i-a}+\frac1{q_i+a},\qquad
0007 \gamma_i=-\frac{p_i-a}{q_i+a}.$$
0008 
0009 在给定的 $(x,y,t)$ 处，记
0010 
0011 $$\theta_i(x,y,t)=S_ix+\omega_it+\ell_i y,\qquad
0012 E_i(x,y,t)=e^{\theta_i(x,y,t)}.$$
0013 
0014 单孤子的解析式为
0015 
0016 $$\begin{aligned}
0017 g(x,y,t)&=1+\frac{E_1(x,y,t)}{S_1},\\
0018 f(x,y,t)&=1+\frac{\gamma_1E_1(x,y,t)}{S_1}.
0019 \end{aligned}$$
0020 
0021 二孤子记
0022 
0023 $$C_{12}=\frac{(p_1-p_2)(q_1-q_2)}
0024 {(p_1+q_1)(p_1+q_2)(p_2+q_1)(p_2+q_2)},$$
0025 
0026 则
0027 
0028 $$\begin{aligned}
0029 g(x,y,t)&=1+\frac{E_1(x,y,t)}{S_1}+\frac{E_2(x,y,t)}{S_2}
0030 +C_{12}E_1(x,y,t)E_2(x,y,t),\\
0031 f(x,y,t)&=1+\frac{\gamma_1E_1(x,y,t)}{S_1}+\frac{\gamma_2E_2(x,y,t)}{S_2}
0032 +\gamma_1\gamma_2C_{12}E_1(x,y,t)E_2(x,y,t).
0033 \end{aligned}$$
0034 
0035 精确解由这两个解析函数直接求导：
0036 
0037 $$\begin{aligned}
0038 u_*(x,y,t)&=2\partial_x[\log f(x,y,t)-\log g(x,y,t)],\\
0039 v_*(x,y,t)&=2\partial_x\partial_y[\log f(x,y,t)+\log g(x,y,t)].
0040 \end{aligned}$$
0041 
0042 展开对数导数，即可直接计算其值：
0043 
0044 $$\begin{aligned}
0045 u_*(x,y,t)&=2\left[\frac{f_x(x,y,t)}{f(x,y,t)}-\frac{g_x(x,y,t)}{g(x,y,t)}\right],\\
0046 v_*(x,y,t)&=2\Bigg[\frac{f_{xy}(x,y,t)}{f(x,y,t)}
0047 -\frac{f_x(x,y,t)f_y(x,y,t)}{f(x,y,t)^2}\\
0048 &\qquad+\frac{g_{xy}(x,y,t)}{g(x,y,t)}
0049 -\frac{g_x(x,y,t)g_y(x,y,t)}{g(x,y,t)^2}\Bigg].
0050 \end{aligned}$$
0051 
0052 这些导数由指数项的解析式得到：$\partial_xE_i=S_iE_i$、$\partial_yE_i=\ell_iE_i$、$\partial_x\partial_yE_i=S_i\ell_iE_i$；交叉项 $E_1E_2$ 的 $x,y$ 指数率分别为 $S_1+S_2$、$\ell_1+\ell_2$。给定 $(x,y,t)$，计算相应的 $f,g$ 及其导数，再代入上式，即得到该点的精确参照值。
0053 
0054 精确参照由 `Exact` 类按上述解析式计算；调用 `uv(js, x, t)`，返回指定空间点与时刻的精确参照值。

0001 class Exact:
0002     def __init__(self, case, h, a=2.):
0003         p, q = map(np.asarray, CASES[case])
0004         self.h = h
0005         s, omega, ell = p+q, q*q-p*p, 1/(p-a)+1/(q+a)
0006         gamma = np.log(-(p-a)/(q+a))
0007         if len(p) == 1:
0008             self.coeff = np.array([1., 1/s[0]])
0009             subsets = np.array([[0.], [1.]])
0010         else:
0011             cross = ((p[0]-p[1])*(q[0]-q[1]) /
0012                 ((p[0]+q[0])*(p[0]+q[1])*(p[1]+q[0])*(p[1]+q[1])))
0013             self.coeff = np.array([1., 1/s[0], 1/s[1], cross])
0014             subsets = np.array([[0., 0.], [1., 0.], [0., 1.], [1., 1.]])
0015         self.rates = subsets @ np.stack((s, omega, ell), axis=1)
0016         self.gamma = subsets @ gamma
0017         self._key, self._cache = None, {}
0018 
0019     def tau(self, js, x, t, is_f=False):
0020         y = (np.atleast_1d(js)+.5)*self.h
0021         logs = (np.log(self.coeff)[:, None, None]
0022             + self.rates[:, 0, None, None]*np.atleast_1d(x)[None, None, :]
0023             + self.rates[:, 1, None, None]*t
0024             + self.rates[:, 2, None, None]*y[None, :, None])
0025         if is_f:
0026             logs = logs + self.gamma[:, None, None]
0027         total = logsumexp(logs, axis=0)
0028         return total, np.exp(logs-total[None, :, :])
0029 
0030     def mean(self, w, k):
0031         return np.einsum('i,ijk->jk', self.rates[:, k], w)
0032 
0033     def cov(self, w, a, b):
0034         return (np.einsum('i,ijk->jk', self.rates[:, a]*self.rates[:, b], w)
0035                 - self.mean(w, a)*self.mean(w, b))
0036 
0037     def uv(self, js, x, t):
0038         key = (float(t), np.asarray(x).tobytes())
0039         if key != self._key:
0040             self._key, self._cache = key, {}
0041         sub = tuple(np.atleast_1d(js))
0042         if sub not in self._cache:
0043             _, f = self.tau(js, x, t, True)
0044             _, g = self.tau(js, x, t)
0045             self._cache[sub] = (2*(self.mean(f, 0)-self.mean(g, 0)),
0046                 2*(self.cov(f, 0, 2)+self.cov(g, 0, 2)))
0047         return tuple(v.copy() for v in self._cache[sub])

0001 class Grid:
0002     def __init__(self, n, L):
0003         self.n, self.L, self.dx = n, L, L/n
0004         self.xi = -L/2 + np.arange(n)*self.dx
0005         ii = np.arange(n)
0006         self.Dxi = coo_matrix((np.concatenate([np.full(n, c/(2*self.dx))
0007             for c in (-1, 1)]), (np.tile(ii, 2),
0008             np.concatenate([(ii+k) % n for k in (-1, 1)]))),
0009             shape=(n, n)).tocsr()
0010         self.Dxxi = coo_matrix((np.concatenate([np.full(n, c/self.dx**2)
0011             for c in (1, -2, 1)]), (np.tile(ii, 3),
0012             np.concatenate([(ii+k) % n for k in (-1, 0, 1)]))),
0013             shape=(n, n)).tocsr()
0014         self.set_s(np.zeros(n))
0015 
0016     def neighbors(self, f, jump=0.):
0017         # f(xi+L)=f(xi)+jump；周期场取 jump=0。
0018         f = np.asarray(f, dtype=np.result_type(f, jump, float))
0019         left, right = np.roll(f, 1, axis=-1), np.roll(f, -1, axis=-1)
0020         left[..., :1] -= jump
0021         right[..., -1:] += jump
0022         return left, right
0023 
0024     def dxi(self, f, jump=0.):
0025         left, right = self.neighbors(f, jump)
0026         return (right-left)/(2*self.dx)
0027 
0028     def dxxi(self, f, jump=0.):
0029         left, right = self.neighbors(f, jump)
0030         return (right-2*f+left)/self.dx**2
0031 
0032     def set_s(self, s):
0033         self.s = np.asarray(s).copy()
0034         self.x, self.J = self.xi+self.s, 1+self.dxi(self.s)
0035         self.Jxi = self.dxi(self.J)
0036         if self.J.min() <= 0 or np.diff(np.r_[self.x, self.x[0]+self.L]).min() <= 0:
0037             raise ValueError('网格 Jacobian 非正或节点交叉')
0038 
0039     def d1(self, f, jump=0.):
0040         return self.dxi(f, jump)/self.J
0041 
0042     def d2(self, f, jump=0.):
0043         return (self.dxxi(f, jump)/self.J**2
0044                 -self.Jxi*self.dxi(f, jump)/self.J**3)
0045 
0046 def delta0(u, ghosts, h):
0047     ext = np.concatenate((ghosts[0][None, :], u, ghosts[1][None, :]))
0048     return (ext[2:]-ext[:-2])/(2*h)
0049 
0050 def ghost_u(G, js, x, t, u):
0051     gl, gr = G.uv([js[0]-1, js[-1]+1], x, t)[0]
0052     # 上侧外推扰动；解析背景与内部数值解分开处理。
0053     e = u[-3:]-G.uv(js[-3:], x, t)[0]
0054     return gl, gr+3*e[-1]-3*e[-2]+e[-3]

0002     def __init__(self, case, model, config):
0003         self.kind, self.h, self.a = model, config['h'], config['a']

0011 
0012     def pack(self, p, w):
0013         return np.r_[p.ravel(), w.ravel()]
0014 
0015     def unpack(self, z):
0016         return z[:self.np].reshape(self.shape[0]-1, -1), z[self.np:].reshape(self.shape)
0017 
0018     def recover_u(self, p, t):
0019         base = self.G.uv([self.js[0]], self.X.x, t)[0][0]
0020         return np.vstack((base, base+self.h*np.cumsum(p, axis=0)))

0001 ### 2.1　SD
0002 
0003 **半离散非线性方程。** 以 $u_j,\omega_j$ 表示第一种非线性形式：
0004 
0005 $$\begin{aligned}
0006 \delta_-u_{j,t}
0007 &+\partial_x\delta_-\left[\frac{u_j^2+\omega_j^2}{2}+2au_j-h\omega_j\right]
0008 +\partial_x^2\left(\delta_-u_j+\frac4hM_-\omega_j\right)=0,\\
0009 \omega_{j,t}&+\partial_x[(u_j+2a)\omega_j-hu_j]-\omega_{j,xx}=0.
0010 \end{aligned}$$
0011 
0012 $\omega_j=\partial_x(\log G_{j+1}-\log G_j)$，物理场为 $v_j=4\omega_j/h+\delta_0u_j$。取演化变量
0013 
0014 $$P_j=\delta_-u_j,\qquad W_j=\frac4h\omega_j=v_j-\delta_0u_j,$$
0015 
0016 并记
0017 
0018 $$H_j=\frac{u_j^2}{2}+2au_j+h^2\left(\frac{W_j^2}{32}-\frac{W_j}{4}\right).$$
0019 
0020 上式成为
0021 
0022 $$\begin{aligned}
0023 P_{j,t}&=-\delta_-\partial_xH_j-\partial_x^2(P_j+M_-W_j),\\
0024 W_{j,t}&=-\partial_x[(u_j+2a)W_j-4u_j]+\partial_x^2W_j.
0025 \end{aligned}$$
0026 
0027 **离散递推。** 每个时间级先由下侧解析边界恢复
0028 
0029 $$u_{j,i}(t)=u_{j_L,i}(t)+h\sum_{k=j_L+1}^{j}P_{k,i}(t),\qquad
0030 v=W+\delta_0u,$$
0031 
0032 再计算
0033 
0034 $$\begin{aligned}
0035 F_P&=-\delta_-D_1H-D_2(P+M_-W),\\
0036 F_W&=-D_1[(u+2a)W-4u]+D_2W,\\
0037 P^{n+1}&=P^n+\Delta t\,F_P(t_n),\qquad
0038 W^{n+1}=W^n+\Delta t\,F_W(t_n).
0039 \end{aligned}$$
0040 
0041 参数：$a=2$、$h=0.125$、$\Delta x=0.15625$、$\Delta t=0.000125$、$T=0.01$。初值为 $P^0=\delta_-u_*(0)$、$W^0=v_*(0)-\delta_0u_*(0)$，$u_{j_L}(t)$ 取解析值。以上为 Euler 更新；同一右端的 RK4 更新见第 3 节。
0042 
0043 **对应代码。** `pt`、`wt` 分别是 $F_P,F_W$；`axis=0` 沿 $y$，`d1,d2` 沿 $x$。

0001 class SDModel(PhysicalModel):
0002     def initial(self):
0003         u, v = self.G.uv(self.js, self.X.x, 0.)
0004         ghosts = self.G.uv([self.js[0]-1, self.js[-1]+1], self.X.x, 0.)[0]
0005         # P = delta_- u, W = 4 omega / h = v - delta_0 u.
0006         return self.pack(np.diff(u, axis=0)/self.h,
0007                          v-delta0(u, ghosts, self.h))
0008 
0009     def fields(self, z, t):
0010         p, w = self.unpack(z)
0011         u = self.recover_u(p, t)
0012         ghosts = ghost_u(self.G, self.js, self.X.x, t, u)
0013         return u, w+delta0(u, ghosts, self.h)
0014 
0015     def rhs(self, t, z):
0016         p, w = self.unpack(z)
0017         u = self.recover_u(p, t)
0018         d1, d2, h, a = self.X.d1, self.X.d2, self.h, self.a
0019         # This is Report (7) after omega = h W / 4.
0020         H = u*u/2+2*a*u+h*h*(w*w/32-w/4)
0021         pt = -np.diff(d1(H), axis=0)/h-d2(p+(w[1:]+w[:-1])/2)
0022         wt = -d1((u+2*a)*w-4*u)+d2(w)
0023         return self.pack(pt, wt)

0001 ### 2.2　SD2
0002 
0003 **原半离散方程。** 以 $Q,R$ 为演化变量：
0004 
0005 $$\begin{aligned}
0006 Q_{j,t}&=-Q_{j,xx}-2aQ_{j,x}-\left[(M_j+M_{j+1})_x+\frac{h^2}{4}(Q_j^2R_j^2-1)\right]Q_j,\\
0007 R_{j,t}&=R_{j,xx}-2aR_{j,x}+\left[(M_j+M_{j+1})_x+\frac{h^2}{4}(Q_j^2R_j^2-1)\right]R_j,\\
0008 M_{j+1}-M_j&=h(1-Q_jR_j).
0009 \end{aligned}$$
0010 
0011 **离散递推。** 令 $S_j=Q_jR_j$、$G_j=h^2(S_j^2-1)/4$、$m_j=M_{j,x}$，下边界确定 $Q_0,Q_{0,t}$，然后计算
0012 
0013 $$\begin{aligned}
0014 m_0&=-\frac{Q_{0,t}+D_2Q_0+2aD_1Q_0}{2Q_0}-\frac{G_0}{2}+\frac h2D_1S_0,\\
0015 m_{j+1}&=m_j-hD_1S_j,\qquad A_j=m_j+m_{j+1},\\
0016 F_Q&=-D_2Q-2aD_1Q-(A+G)Q,\\
0017 F_R&=D_2R-2aD_1R+(A+G)R,\\
0018 Q_j^{n+1}&=Q_j^n+\Delta t\,F_{Q,j}(t_n)\quad(j\ge1),\qquad
0019 R_j^{n+1}=R_j^n+\Delta t\,F_{R,j}(t_n)\quad(j\ge0),\\
0020 u&=2(D_1Q)/Q,\qquad v=4(1-QR)+\delta_0u.
0021 \end{aligned}$$
0022 
0023 参数：$a=2$、$h=0.125$、$\Delta x=0.15625$、$\Delta t=0.000125$、$T=0.01$。初值与下边界通过
0024 
0025 $$D_1Q_j^0=\tfrac12u_j^0Q_j^0,\qquad Q_{j,0}^0=1,\qquad
0026 R_j^0=\frac{1-(v_j^0-\delta_0u_j^0)/4}{Q_j^0}$$
0027 
0028 确定。下边界每个时间级求解 $D_1Q_0(t)=u_{*,0}(t)Q_0(t)/2$、$Q_{0,0}(t)=1$；这里的 $D_1$ 使用代码 `dx` 的端点跃变量修正。内部 $Q$ 与全部 $R$ 按上述右端推进，Euler 与 RK4 共用此右端。

0001 class SD2Lift(PhysicalModel):
0002     def __init__(self, case, config):
0003         super().__init__(case, 'SD2', config)
0004         n, dx = self.X.n, self.X.dx
0005         self.D = self.X.Dxi.tocsc()
0006         self.b = np.zeros(n)
0007         self.b[0] = self.b[-1] = 1/(2*dx)
0008         self.norm = csc_matrix(([1.], ([0], [0])), shape=(1, n))
0009         self.initial_right, self._boundary_key = None, None
0010         self.jump = self.rjump = 0.
0011 
0012     def dx(self, f, jump=0.):
0013         return self.X.d1(f, jump)
0014 
0015     def lift(self, u):
0016         A = bmat([[self.D-diags(.5*self.X.J*u), csc_matrix(self.b[:, None])],
0017                   [self.norm, csc_matrix((1, 1))]], format='csc')
0018         lu = splu(A)
0019         sol = lu.solve(np.r_[np.zeros(self.X.n), 1.])
0020         q, jump = sol[:-1], sol[-1]
0021         if q.min() <= 0 or 1+jump <= 0:
0022             raise ValueError('SD2 提升后的 Q 或端点比值非正')
0023         return q, jump, lu

0025     def boundary(self, t):
0026         key = (float(t), self.X.x.tobytes(), self.X.J.tobytes())
0027         if key != self._boundary_key:
0028             _, f = self.G.tau([self.js[0]], self.X.x, t, True)
0029             _, g = self.G.tau([self.js[0]], self.X.x, t)
0030             u = 2*(self.G.mean(f, 0)-self.G.mean(g, 0))[0]
0031             ut = 2*(self.G.cov(f, 0, 1)-self.G.cov(g, 0, 1))[0]
0032             ux = 2*(self.G.cov(f, 0, 0)-self.G.cov(g, 0, 0))[0]
0033             q, jump, lu = self.lift(u)
0034             self._boundary_key = key
0035             self._boundary_value = q, jump, lu, u, ut, ux
0036         q, jump, lu, u, ut, ux = self._boundary_value
0037         if self.initial_right is not None:
0038             right = self.initial_right*(1+jump)/self.initial_right[0]
0039             self.jump, self.rjump = (right-1)[:, None], (1/right-1)[:, None]
0040         return self._boundary_value

0025         H = self.h**2/4*(w*w-1)

0027         mx0 = (A0+self.h*wx[0])/2

0001 ### 2.3　FD
0002 
0003 **原连续方程。**
0004 
0005 $$\begin{aligned}
0006 u_{yt}&=-\partial_x[(u+2a)u_y]-v_{xx},\\
0007 v_t&=-\partial_x[(u+2a)v-4u]-u_{xxy}.
0008 \end{aligned}$$
0009 
0010 **离散递推。** 取 $P=\delta_-u$，每个时间级由 $u_{j,i}=u_{j-1,i}+hP_{j,i}$ 恢复 $u$，然后计算
0011 
0012 $$\begin{aligned}
0013 F_P&=-\delta_-D_1\left(\frac{u^2}{2}+2au\right)-D_2M_-v,\\
0014 F_v&=-D_1[(u+2a)v-4u]-D_2\delta_0u,\\
0015 P^{n+1}&=P^n+\Delta t\,F_P(t_n),\qquad v^{n+1}=v^n+\Delta t\,F_v(t_n).
0016 \end{aligned}$$
0017 
0018 参数：$a=2$、$h=0.125$、$\Delta x=0.15625$、$\Delta t=0.000125$、$T=0.01$。初值为 $P^0=\delta_-u_*(0)$、$v^0=v_*(0)$。Euler 与 RK4 共用此右端。
0019 
0020 **对应代码。** `fp`、`fv` 分别是 $F_P,F_v$，`uy` 是 $\delta_0u$。

0001 class FDModel(PhysicalModel):
0002     def rhs(self, t, z):
0003         p, v = self.unpack(z)
0004         u = self.recover_u(p, t)
0005         ghosts = ghost_u(self.G, self.js, self.X.x, t, u)

0008         fp = -np.diff(d1(u*u/2+2*a*u), axis=0)/h-d2((v[1:]+v[:-1])/2)
0009         fv = -d1((u+2*a)*v)-d2(uy)+4*d1(u)

0001 ### 2.4　网格策略
0002 
0003 固定网格取 $s=0$、$V=0$。动网格初始节点按 $\bar\rho$ 的累积积分等分，之后使用
0004 
0005 $$\begin{aligned}
0006 x_i&=\xi_i+s_i,\qquad J_i=1+D_\xi s_i,\qquad D_1=J^{-1}D_\xi,\\
0007 D_2f&=J^{-2}D_{\xi\xi}f-J^{-3}(D_\xi J)(D_\xi f),\\
0008 \rho_j&=1-(v_j-\delta_0u_j)/4,\qquad
0009 \bar\rho=\frac1{24}\sum_j\rho_j,\\
0010 \bar q&=\frac1{24}\sum_j[(u_j+2a)\rho_j-D_1\rho_j-2a],\qquad
0011 V_i=\frac{\bar q_i-\bar q_0}{\bar\rho_i},\\
0012 \dot x_i&=V_i,\qquad \dot z=\mathcal F(t,z)+VD_1z.
0013 \end{aligned}$$

0001 ## 3　时间推进与误差评价
0002 
0003 三种方案分别以第 2 节的离散右端 $\mathcal F$ 推进。Euler 更新为 $z^{n+1}=z^n+\Delta t\,\mathcal F(t_n,z^n)$；RK4 的递推为
0004 
0005 $$\begin{aligned}
0006 k_1&=\mathcal F(t_n,z^n),\\
0007 k_2&=\mathcal F\left(t_n+\frac{\Delta t}{2},z^n+\frac{\Delta t}{2}k_1\right),\\
0008 k_3&=\mathcal F\left(t_n+\frac{\Delta t}{2},z^n+\frac{\Delta t}{2}k_2\right),\\
0009 k_4&=\mathcal F(t_n+\Delta t,z^n+\Delta t k_3),\\
0010 z^{n+1}&=z^n+\frac{\Delta t}{6}(k_1+2k_2+2k_3+k_4).
0011 \end{aligned}$$
0012 
0013 状态与节点使用同一时间算法同步更新。
0014 
0015 在 $x\in[-10,10]$ 的 4001 个等距点及全部 $y$ 层上，三次样条重构数值场，计算
0016 
0017 $$E_f(T)=\max_{(x,y)\in\mathcal G}|f_h(x,y,T)-f_*(x,y,T)|,\qquad f\in\{u,v\}.$$
0018 
0019 三种方案从相同的物理初值 $u_*(0),v_*(0)$ 出发。
0020 
0021 时间算法比较另加入 Crank–Nicolson（C–N）：
0022 
0023 $$z^{n+1}=z^n+\frac{\Delta t}{2}\left[\mathcal F(t_n,z^n)+\mathcal F(t_{n+1},z^{n+1})\right].$$
0024 
0025 这是二阶隐式梯形法，采用迭代求解并检查隐式方程残差。

0001 def step(fun, t, state, dt, method):
0002     if method == 'Euler':
0003         return state+dt*fun(t, state)
0004     if method == 'CN':
0005         # 隐式梯形法：两端右端的平均，不是中点状态的右端。
0006         f0 = fun(t, state)
0007         candidate = state+dt*f0
0008         tolerance = 1e-12+1e-11*max(1., float(np.max(abs(state))))
0009         for iteration in range(1, 81):
0010             residual = candidate-state-dt*(f0+fun(t+dt, candidate))/2
0011             residual_max = float(np.max(abs(residual)))
0012             if residual_max <= tolerance:
0013                 step.cn_last = dict(iterations=iteration, residual=residual_max, tolerance=tolerance)
0014                 return candidate
0015             candidate = candidate-residual
0016         raise ValueError(f'Crank–Nicolson 隐式迭代未收敛：残差 {residual_max:.3e}')
0017     if method != 'RK4':
0018         raise ValueError('时间算法应为 Euler、RK4 或 CN')
0019     k1 = fun(t, state)
0020     k2 = fun(t+dt/2, state+dt*k1/2)
0021     k3 = fun(t+dt/2, state+dt*k2/2)
0022     k4 = fun(t+dt, state+dt*k3)
0023     return state+dt*(k1+2*k2+2*k3+k4)/6

0004 def checked_window(window, available, label):
0005     lo, hi = window
0006     if not np.isfinite([lo, hi]).all() or lo >= hi:
0007         raise ValueError(f'{label} 必须是有限且递增的 (下限, 上限)。')
0008     if lo < available[0]-1e-10 or hi > available[1]+1e-10:
0009         raise ValueError(f'{label}={window} 超出已计算范围 {available}；请扩大计算域后重新准备场数据。')
0010     return lo, hi
