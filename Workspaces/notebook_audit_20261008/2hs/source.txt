0001 def hs_exact_X(X, t):
0002     """式（2）的解析场；X 可以是标量或数组。"""
0003     X = np.asarray(X, dtype=float)
0004     z = np.tanh((3.75 * X - 0.6 * t) / 2)
0005     S = 1 - z * z
0006     zX = 1.875 * S
0007     zXX = -2 * zX * 1.875 * z
0008     J, JX = 1 + 0.3 * zX, 0.3 * zXX
0009     u, uX = 0.09 * S, -0.18 * z * zX
0010     uXX = -0.18 * (zX * zX + z * zXX)
0011     # 链式法则：d/dx=(1/J)d/dX，其中 J=dx/dX。
0012     uxx = (uXX * J - uX * JX) / (J * J * J)
0013     return dict(x=X - 0.5 + 0.3 * z, u=u, rho=1 / J, m=uxx + 2, X=X)
0014 
0015 def hs_exact_physical(x, t):
0016     """利用 dx/dX>0，在 [x+0.2,x+0.8] 中二分反解 X。"""
0017     x = np.asarray(x, dtype=float)
0018     lo, hi = x + 0.2, x + 0.8
0019     for _ in range(60):
0020         mid = (lo + hi) / 2
0021         lower = hs_exact_X(mid, t)["x"] < x
0022         lo, hi = np.where(lower, mid, lo), np.where(lower, hi, mid)
0023     value = hs_exact_X((lo + hi) / 2, t)
0024     if np.max(np.abs(value["x"] - x)) > 2e-12:
0025         raise ArithmeticError("物理坐标反解未达到容差。")
0026     return value

0001 ## 2　空间离散
0002 
0003 ### 2.1　Integrable
0004 
0005 原连续方程为
0006 
0007 $$u_{tx}-uu_{xx}-\frac12u_x^2-4u-\frac12(\rho^2-1)=0,
0008 \qquad \rho_t=(u\rho)_x.$$
0009 
0010 将辅助坐标 $X\in[-6,6]$ 等分为 $N=2400$ 段，
0011 $X_k=-6+ka$、$a=0.005$；式（2）将 $X=-6,6$ 映为初始物理端点，
0012 区间约为 $[-6.8,\,5.8]$。
0013 
0014 令 $d_k=x_{k+1}-x_k$、$b_k=u_{k+1}-u_k$。$c=1$ 时的半离散方程为
0015 
0016 $$\begin{aligned}
0017 \dot d_k&=-b_k,\qquad \dot x_0=-u_0(t),\\
0018 \dot b_k&=2d_k(u_{k+1}+u_k)
0019 +\frac{[d_k^2-a(a-1)]^2-b_k^2}{2d_k}-\frac{d_k}{2}.
0020 \end{aligned}\tag{3}$$
0021 
0022 取解析左端值 $u_0(t)$，节点及密度按下式递推：
0023 
0024 $$u_{k+1}=u_k+b_k,\qquad x_{k+1}=x_k+d_k,\qquad
0025 \rho_{k+1/2}=\frac a{d_k}.\tag{4}$$
0026 
0027 参数：$c=1$、$a=0.005$、$N=2400$、$\Delta t=0.003125$、$T=0.5$。
0028 以下代码实现式（4）的恢复与式（3）的右端；RK4 时间递推见第 3 节。


=== CELL 8 code ===
0001 def hs_integrable_fields(t, state):
0002     # 状态：[b_0,...,b_(N-1), d_0,...,d_(N-1), x_0]。
0003     b, d = state[:hs_N], state[hs_N:2 * hs_N]
0004     if not np.isfinite(state).all() or np.any(d <= 0):
0005         raise ArithmeticError("Integrable 出现非有限状态或非正胞元格距。")
0006     u0 = hs_exact_X(hs_Xleft, t)["u"]
0007     u = np.r_[u0, u0 + np.cumsum(b)]
0008     x = np.r_[state[-1], state[-1] + np.cumsum(d)]
0009     return dict(x=x, u=u, rhoX=(x[:-1] + x[1:]) / 2, rho=hs_a / d)
0010 
0011 def hs_integrable_rhs(t, state):
0012     # 式（3）：b、d 和 x_0 的时间导数。
0013     u = hs_integrable_fields(t, state)["u"]
0014     b, d = state[:hs_N], state[hs_N:2 * hs_N]
0015     C = d * d - hs_a * (hs_a - 1)
0016     db = 2 * d * (u[1:] + u[:-1]) + (C * C - b * b) / (2 * d) - d / 2
0017     return np.r_[db, -b, -u[0]]

0001 ### 2.2　FD
0002 
0003 对式（1）作 $x$ 微分，令 $m=u_{xx}+2$，原方程写成
0004 
0005 $$m_t=um_x+2mu_x+\rho\rho_x,\qquad
0006 \rho_t=u\rho_x+\rho u_x,\qquad u_{xx}=m-2.$$
0007 
0008 在初始物理端点之间等分 2400 段，$x_i=x_L+i\Delta x$；
0009 物理端点由解析坐标映射在 $X=-6,6$ 处给出，
0010 单孤子、二孤子的区间分别约为 $[-6.8,\,5.8]$、$[-7.7,\,5.7]$。
0011 
0012 中心差分及离散方程为
0013 
0014 $$(D_1f)_i=\frac{f_{i+1}-f_{i-1}}{2\Delta x},\qquad
0015 (D_2f)_i=\frac{f_{i+1}-2f_i+f_{i-1}}{\Delta x^2}.\tag{5}$$
0016 
0017 $$\begin{aligned}
0018 \dot m_i&=u_i(D_1m)_i+2m_i(D_1u)_i+\rho_i(D_1\rho)_i,\\
0019 \dot\rho_i&=u_i(D_1\rho)_i+\rho_i(D_1u)_i,\\
0020 u_{i+1}-2u_i+u_{i-1}&=\Delta x^2(m_i-2).
0021 \end{aligned}\tag{6}$$
0022 
0023 参数：$c=1$、$N=2400$、$\Delta t=0.003125$、$T=0.5$；
0024 $\Delta x\approx0.005250$（单孤子）或 $\Delta x\approx0.005591$（二孤子）。
0025 两端取解析值。以下代码先联立求出 $u$，再计算式（6）的时间导数。


=== CELL 10 code ===
0001 def hs_recover_u(m, ub_left, ub_right):
0002     # Thomas 消元：u_(i-1)-2u_i+u_(i+1)=dx^2*(m_i-2)。
0003     count = hs_N - 1
0004     c, d = np.zeros(count), np.zeros(count)
0005     for j in range(count):
0006         diagonal = -2 - (c[j - 1] if j else 0)
0007         rhs = hs_dx * hs_dx * (m[j + 1] - 2)
0008         if j == 0:
0009             rhs -= ub_left
0010         if j == count - 1:
0011             rhs -= ub_right
0012         c[j] = 1 / diagonal if j < count - 1 else 0
0013         d[j] = (rhs - (d[j - 1] if j else 0)) / diagonal
0014     u = np.empty(hs_N + 1)
0015     u[0], u[-1] = ub_left, ub_right
0016     for j in range(count - 1, -1, -1):
0017         u[j + 1] = d[j] - (c[j] * u[j + 2] if j < count - 1 else 0)
0018     return u
0019 
0020 def hs_fd_fields(t, state):
0021     # FD 状态为内部 m 与 rho，端点取解析边界。
0022     boundary = hs_exact_physical(np.array([hs_left, hs_right]), t)
0023     m = np.r_[boundary["m"][0], state[:hs_N - 1], boundary["m"][1]]
0024     rho = np.r_[boundary["rho"][0], state[hs_N - 1:], boundary["rho"][1]]
0025     if not np.isfinite(m).all() or not np.isfinite(rho).all() or np.any(rho <= 0):
0026         raise ArithmeticError("FD 出现非有限状态或非正密度。")
0027     u = hs_recover_u(m, *boundary["u"])
0028     return dict(x=hs_fixed_x, u=u, rhoX=hs_fixed_x, rho=rho, m=m)
0029 
0030 def hs_fd_rhs(t, state):
0031     # 式（6）：固定网格上的一阶中心差分。
0032     fields = hs_fd_fields(t, state)
0033     m, rho, u = (fields[k] for k in ("m", "rho", "u"))
0034     ux = (u[2:] - u[:-2]) / (2 * hs_dx)
0035     mx = (m[2:] - m[:-2]) / (2 * hs_dx)
0036     rx = (rho[2:] - rho[:-2]) / (2 * hs_dx)
0037     return np.r_[u[1:-1] * mx + 2 * m[1:-1] * ux + rho[1:-1] * rx,
0038                  u[1:-1] * rx + rho[1:-1] * ux]

0001 ## 3　时间推进
0002 
0003 记空间系统为 $\dot z=\mathcal F(t,z)$，时间递推采用 RK4：
0004 
0005 $$\begin{aligned}
0006 K_1&=\mathcal F(t_n,z^n),&
0007 K_2&=\mathcal F\!\left(t_n+\frac{\Delta t}{2},z^n+\frac{\Delta t}{2}K_1\right),\\
0008 K_3&=\mathcal F\!\left(t_n+\frac{\Delta t}{2},z^n+\frac{\Delta t}{2}K_2\right),&
0009 K_4&=\mathcal F(t_n+\Delta t,z^n+\Delta tK_3),\\
0010 z^{n+1}&=z^n+\frac{\Delta t}{6}(K_1+2K_2+2K_3+K_4).
0011 \end{aligned}\tag{7}$$
0012 
0013 取 $\Delta t=0.003125$、$T=0.5$，共 160 步。每个阶段按式（4）或（6）恢复场值。


=== CELL 14 code ===
0001 def hs_rk4_step(rhs, t, state, delta):
0002     # 式（7）：每一级重新恢复场值和解析边界。
0003     k1 = rhs(t, state)
0004     k2 = rhs(t + delta / 2, state + delta / 2 * k1)
0005     k3 = rhs(t + delta / 2, state + delta / 2 * k2)
0006     k4 = rhs(t + delta, state + delta * k3)
0007     return state + delta * (k1 + 2 * k2 + 2 * k3 + k4) / 6

0001 def hs_interpolate(xs, values, targets):
0002     # 物理节点上的分段线性重构。
0003     if targets[0] < xs[0] or targets[-1] > xs[-1]:
0004         raise ValueError("数值网格未覆盖整个评价区间。")
0005     # side='left' 与原报告在恰好落于节点时采用相同的左区间。
0006     j = np.clip(np.searchsorted(xs, targets, side="left") - 1, 0, len(xs) - 2)
0007     weight = (targets - xs[j]) / (xs[j + 1] - xs[j])
0008     return values[j] + weight * (values[j + 1] - values[j])

0001 ## 5　二孤子
0002 
0003 取 $p_1=1.1$、$p_2=1.25$、$q_1=11$、$q_2=5$、$c=1$，相位常数均为 $\log6.5$。
0004 连续二孤子的 $\tau$ 函数为
0005 
0006 $$\begin{aligned}
0007 f&=1+e^{\theta_1}+e^{\theta_2}+A_{12}e^{\theta_1+\theta_2},
0008 \qquad A_{12}=\frac{12}{507},\\
0009 \theta_i&=(p_i-q_i)X+\left(\frac1{p_i}-\frac1{q_i}\right)t+\log6.5.
0010 \end{aligned}\tag{9}$$
0011 
0012 相应的物理坐标与场为
0013 
0014 $$u=\partial_t^2\log f,\qquad
0015 x=X-\partial_t\log f-\frac1{q_1}-\frac1{q_2},\qquad
0016 \rho=\frac1{1-\partial_X\partial_t\log f}.\tag{10}$$
0017 
0018 归一化后的四项指数构成权重。$u$ 是时间指数系数的方差，
0019 $J=\partial x/\partial X$ 由时间、空间指数系数的协方差给出；
0020 进一步的混合导数确定 $m=u_{xx}+2$，供 FD 边界使用。

0015 def hs_two_exact_X(X, t):
0016     """u、J=dx/dX 和 m=u_xx+2 均由 τ 的解析导数计算。"""
0017     X = np.asarray(X, dtype=float)
0018     flat = X.reshape(-1)
0019     logs = hs_two_beta*flat + hs_two_omega*t + hs_two_logcoef
0020     # 减去列最大值避免指数溢出，不改变归一化权重。
0021     weights = np.exp(logs - np.max(logs, axis=0))
0022     weights /= np.sum(weights, axis=0)
0023     mean_o = np.sum(weights*hs_two_omega, axis=0)
0024     mean_b = np.sum(weights*hs_two_beta, axis=0)
0025     do, db = hs_two_omega-mean_o, hs_two_beta-mean_b
0026     u = np.sum(weights*do**2, axis=0)
0027     cov = np.sum(weights*do*db, axis=0)
0028     J = 1-cov
0029     JX = -np.sum(weights*do*db**2, axis=0)
0030     uX = np.sum(weights*do**2*db, axis=0)
0031     uXX = (np.sum(weights*do**2*db**2, axis=0)
0032            - u*np.sum(weights*db**2, axis=0) - 2*cov**2)
0033     if not np.isfinite(J).all() or np.any(J <= 0):
0034         raise ArithmeticError("二孤子物理坐标要求 J=dx/dX>0。")
0035     uxx = (uXX*J-uX*JX)/J**3
0036     return {k: value.reshape(X.shape) for k, value in dict(
0037         x=flat-mean_o-hs_two_shift, u=u, rho=1/J, m=uxx+2,
0038         X=flat, J=J, uX=uX, uXX=uXX, JX=JX).items()}
0039 
0040 def hs_two_exact_physical(x, t):
0041     """由 τ 权重的凸组合界定 X 的区间，再二分反解。"""
0042     x = np.asarray(x, dtype=float)
0043     lo = x + hs_two_shift + float(np.min(hs_two_omega))
0044     hi = x + hs_two_shift + float(np.max(hs_two_omega))
0045     for _ in range(60):
0046         mid = (lo+hi)/2
0047         lower = hs_two_exact_X(mid, t)["x"] < x
0048         lo, hi = np.where(lower, mid, lo), np.where(lower, hi, mid)
0049     value = hs_two_exact_X((lo+hi)/2, t)
0050     if np.max(np.abs(value["x"]-x)) > 2e-12:
0051         raise ArithmeticError("二孤子物理坐标反解未达到容差。")
0052     return value
